# 5. 生活等级完整实现

## 5.1 等级定义

```ts
export const LIVING_LEVELS = {
  L1: {
    label: "极简生存",
    description: "只覆盖基础吃住交通，几乎没有娱乐和弹性消费。",
    multiplier: 0.55,
  },
  L2: {
    label: "基础幸福生活",
    description: "低欲望生活，能覆盖基本生活并保留少量弹性。",
    multiplier: 0.75,
  },
  L3: {
    label: "普通稳定",
    description: "正常租房、餐饮、交通和少量娱乐支出。",
    multiplier: 1,
  },
  L4: {
    label: "舒适生活",
    description: "居住、饮食、社交和兴趣支出更充足。",
    multiplier: 1.35,
  },
  L5: {
    label: "富足自由",
    description: "高生活质量，有较强消费自由度。",
    multiplier: 1.8,
  },
} as const;
```

## 5.2 根据支出判断等级

```ts
export function calculateLivingLevel(
  monthlyExpense: number,
  city: CityCostProfile
): LivingLevel {
  const ratio = monthlyExpense / city.baseMonthlyCost;

  if (ratio < 0.6) return "L1";
  if (ratio < 0.85) return "L2";
  if (ratio < 1.2) return "L3";
  if (ratio < 1.6) return "L4";
  return "L5";
}
```

## 5.3 根据等级计算推荐支出

```ts
export function getMonthlyCostByLivingLevel(
  level: LivingLevel,
  city: CityCostProfile
): number {
  return Math.round(city.baseMonthlyCost * LIVING_LEVELS[level].multiplier);
}
```

## 5.4 当前 vs 目标生活等级对比

```ts
export function compareLivingLevels(
  current: LivingLevel,
  target: LivingLevel
) {
  const order: LivingLevel[] = ["L1", "L2", "L3", "L4", "L5"];
  const currentIndex = order.indexOf(current);
  const targetIndex = order.indexOf(target);

  return {
    diff: targetIndex - currentIndex,
    isUpgrade: targetIndex > currentIndex,
    isDowngrade: targetIndex < currentIndex,
    isSame: targetIndex === currentIndex,
  };
}
```

# 6. 核心计算引擎

## 6.1 年度模拟函数

```ts
export function simulateYearly(
  input: CalculatorFormState,
  city: CityCostProfile
): YearlySimulationResult[] {
  const results: YearlySimulationResult[] = [];

  let balance = input.currentSavings || 0;
  let monthlySalary = input.monthlySalaryIncome || 0;
  let monthlyPassiveIncome = input.monthlyPassiveIncome || 0;
  let monthlyExpense = input.monthlyExpense || 0;

  for (let age = input.currentAge; age <= input.maxAge; age++) {
    const yearIndex = age - input.currentAge;
    const startBalance = balance;

    const hasSalary =
      input.incomeType === "with_salary" &&
      (!input.salaryStopAge || age < input.salaryStopAge);

    const salaryIncome = hasSalary ? monthlySalary * 12 : 0;
    const passiveIncome = monthlyPassiveIncome * 12;
    const totalIncome = salaryIncome + passiveIncome;
    const expense = monthlyExpense * 12;

    const investmentReturn = Math.max(balance, 0) * input.annualReturnRate;
    const endBalance = balance + investmentReturn + totalIncome - expense;

    const livingLevel = calculateLivingLevel(monthlyExpense, city);

    results.push({
      age,
      yearIndex,
      startBalance,
      salaryIncome,
      passiveIncome,
      totalIncome,
      expense,
      investmentReturn,
      endBalance,
      monthlyExpense,
      livingLevel,
      isDepleted: endBalance < 0,
    });

    balance = endBalance;

    if (balance < 0) {
      break;
    }

    monthlySalary = monthlySalary * (1 + input.annualIncomeGrowthRate);
    monthlyPassiveIncome = monthlyPassiveIncome * (1 + input.annualIncomeGrowthRate);
    monthlyExpense = monthlyExpense * (1 + input.annualExpenseGrowthRate);
  }

  return results;
}
```

## 6.2 计算可持续年龄

```ts
export function calculateSustainableAge(
  input: CalculatorFormState,
  yearlyResults: YearlySimulationResult[]
) {
  const depleted = yearlyResults.find((item) => item.isDepleted);

  if (depleted) {
    return {
      sustainableAge: depleted.age,
      depletionAge: depleted.age,
      yearsSustainable: depleted.age - input.currentAge,
    };
  }

  const last = yearlyResults[yearlyResults.length - 1];

  return {
    sustainableAge: last.age,
    depletionAge: undefined,
    yearsSustainable: last.age - input.currentAge,
  };
}
```

## 6.3 计算月结余

```ts
export function calculateMonthlySurplus(input: CalculatorFormState) {
  return (
    input.monthlySalaryIncome +
    input.monthlyPassiveIncome -
    input.monthlyExpense
  );
}
```

## 6.4 计算储蓄率

```ts
export function calculateSavingsRate(input: CalculatorFormState) {
  if (input.monthlySalaryIncome <= 0) return undefined;

  return calculateMonthlySurplus(input) / input.monthlySalaryIncome;
}
```

## 6.5 模拟到目标年龄

```ts
export function simulateToTargetAge(
  input: CalculatorFormState,
  city: CityCostProfile,
  targetAge: number
) {
  const results = simulateYearly(
    {
      ...input,
      maxAge: targetAge,
    },
    city
  );

  const last = results[results.length - 1];

  return {
    results,
    finalBalance: last?.endBalance ?? input.currentSavings,
    isSustainableToTarget: Boolean(last && last.age >= targetAge && last.endBalance >= 0),
  };
}
```

export function calculateTargetWithSalary(
  input: CalculatorFormState,
  city: CityCostProfile
): CalculatorResult {
  if (!input.targetAge) {
    throw new Error("目标年龄不能为空");
  }

  const currentLivingLevel = calculateLivingLevel(input.monthlyExpense, city);

  const targetMonthlyExpense = input.targetLivingLevel
    ? getMonthlyCostByLivingLevel(input.targetLivingLevel, city)
    : input.monthlyExpense;

  const targetInput = {
    ...input,
    monthlyExpense: targetMonthlyExpense,
  };

  const { results, finalBalance, isSustainableToTarget } = simulateToTargetAge(
    targetInput,
    city,
    input.targetAge
  );

  const sustainability = calculateSustainableAge(targetInput, results);
  const fundingGap = finalBalance < 0 ? Math.abs(finalBalance) : 0;

  const monthlySurplus = calculateMonthlySurplus(targetInput);
  const annualSurplus = monthlySurplus * 12;
  const savingsRate = calculateSavingsRate(targetInput);

  const risk = calculateRiskLevel(targetInput, results, city);

  const suggestions = generateTargetProjectionSuggestions(
    targetInput,
    city,
    input.targetAge,
    isSustainableToTarget
  );

  const affordableLivingLevelAtTarget = calculateAffordableLivingLevelAtTarget(
    targetInput,
    city,
    input.targetAge
  );

  return {
    mode: "target_projection",
    incomeType: "with_salary",
    currentAge: input.currentAge,
    targetAge: input.targetAge,
    sustainableAge: sustainability.sustainableAge,
    yearsSustainable: sustainability.yearsSustainable,
    depletionAge: sustainability.depletionAge,
    isSustainableToTarget,
    finalBalance,
    fundingGap,
    currentLivingLevel,
    targetLivingLevel: input.targetLivingLevel,
    affordableLivingLevelAtTarget,
    monthlySurplus,
    annualSurplus,
    savingsRate,
    riskLevel: risk.riskLevel,
    riskReasons: risk.reasons,
    yearlyResults: results,
    suggestions,

## 8.2 二分搜索工具函数

```ts
function binarySearchMinimum(
  min: number,
  max: number,
  checker: (value: number) => boolean,
  precision = 1
): number {
  let left = min;
  let right = max;
  let answer = max;

  while (right - left > precision) {
    const mid = Math.floor((left + right) / 2);

    if (checker(mid)) {
      answer = mid;
      right = mid;
    } else {
      left = mid;
    }
  }

  return answer;
}
```

    additionalSavingsRequired: additionalSavings,
    priority: 3,
    difficulty: "hard",
  });

  const passiveIncomeIncrease = findRequiredPassiveIncomeIncrease(input, city, targetAge);

    description: `每月增加约 ${passiveIncomeIncrease} 元被动收入，可以帮助你维持到 ${targetAge} 岁。`,

# 9. 风险等级完整实现

## 9.1 风险规则

```text
critical：目标不可达，或 5 年内资金耗尽
high：目标可达但最终余额不足 12 个月支出
medium：最终余额为 12-36 个月支出
low：最终余额大于 36 个月支出
```

收益率修正：

```text
年化收益率 > 8%，风险上调一级
月结余 < 0，风险上调一级
支出增长率 > 收入增长率，风险上调一级
```

## 9.2 实现

```ts
export function calculateRiskLevel(
  input: CalculatorFormState,
  yearlyResults: YearlySimulationResult[],
  city: CityCostProfile
): { riskLevel: RiskLevel; reasons: string[] } {
  const reasons: string[] = [];
  const last = yearlyResults[yearlyResults.length - 1];
  const depleted = yearlyResults.find((item) => item.isDepleted);

  let score = 0;

  if (depleted) {
    const yearsToDepletion = depleted.age - input.currentAge;

    if (yearsToDepletion <= 5) {
      score += 4;
      reasons.push("资金可能在 5 年内耗尽");
    } else {
      score += 3;
      reasons.push("模拟周期内资金会耗尽");
    }
  }

  if (last && last.endBalance >= 0) {
    const annualExpense = input.monthlyExpense * 12;
    const safetyYears = annualExpense > 0 ? last.endBalance / annualExpense : 0;

    if (safetyYears < 1) {
      score += 3;
      reasons.push("最终资金安全垫不足 1 年支出");
    } else if (safetyYears < 3) {
      score += 2;
      reasons.push("最终资金安全垫不足 3 年支出");
    } else {
      reasons.push("资金安全垫相对充足");
    }
  }

  const monthlySurplus = calculateMonthlySurplus(input);

  if (monthlySurplus < 0) {
    score += 1;
    reasons.push("当前每月现金流为负，需要消耗存款");
  }

  if (input.annualReturnRate > 0.08) {
    score += 1;
    reasons.push("收益率假设较高，结果可能偏乐观");
  }

  if (input.annualExpenseGrowthRate > input.annualIncomeGrowthRate) {
    score += 1;
    reasons.push("支出增长率高于收入增长率");
  }

  if (score >= 4) return { riskLevel: "critical", reasons };
  if (score >= 3) return { riskLevel: "high", reasons };
  if (score >= 1) return { riskLevel: "medium", reasons };
  return { riskLevel: "low", reasons };
}
```

export function calculateResult(

  city: CityCostProfile

# 24. 必须避免的问题

## 24.1 不要把结果说死

错误：

```text
你一定可以幸福生活到 60 岁。
```

正确：

```text
在当前参数假设下，你可以维持到 60 岁。实际结果会受到收入、支出、收益率和突发事件影响。
```

## 24.2 不要制造焦虑

错误：

```text
你的财务状况非常危险。
```

正确：

```text
当前方案存在明显缺口，建议优先关注支出控制和收入补充。
```

## 24.3 不要把高收益率当默认

默认收益率建议：

```text
保守：2%
稳健：4%
积极：6%
自定义
```

超过 8% 必须提示风险。

# 26. 最终开发目标

这不是一个简单计算器，而是一个完整的“生活可持续性模拟器”。

完整首版应该做到：

```text
用户能回答：我现在这样能撑到几岁？
用户能回答：我想撑到某个年龄，需要怎么调？
用户能看到：当前生活等级和目标生活等级差距。
用户能看到：钱会在哪一年变多、变少、耗尽。
用户能操作：拖动收入、支出、收益率，实时看结果变化。
用户能保存：多个方案。
用户能对比：哪种方案更可持续。
用户能分享：脱敏后的结果。
```

这就是流程图上所有功能的全量实现版本。
